Let's use reflection and expression rewriting to apply the tenant config to all ITenanted entities automatically.

Keeping it DRY (with magic 🪄)


protected override void OnModelCreating(ModelBuilder builder)
{
    var tenantedModels = builder.Model.GetEntityTypes()
        .Where(e =>
            typeof(ITenanted).IsAssignableFrom(e.GetType()));
    foreach (var model in tenantedModels)
    {
        builder.Entity(model.ClrType)
            .HasQueryFilter((ITenanted e) =>
                e.TenantId == Tenant.Id)
            .HasIndex(nameof(ITenanted.TenantId));
        builder.Entity(model.ClrType)
            .Property(nameof(ITenanted.TenantId))
            .IsRequired()
            .HasValueGenerator<TenantIdValueGenerator>();
    }
    //...
}
